from decimal import Decimal

from django.conf import settings
from django.core.validators import MinValueValidator
from django.db import models


class Card(models.Model):
    """Human-readable work item codes are imported from a team's existing tracker."""

    class Status(models.TextChoices):
        OPEN = "open", "Open"
        ACTIVE = "active", "In progress"
        DONE = "done", "Done"

    code = models.CharField(primary_key=True, max_length=40)
    owner = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE)
    title = models.CharField(max_length=160)
    status = models.CharField(max_length=12, choices=Status.choices, default=Status.OPEN)
    estimate = models.DecimalField(
        max_digits=9,
        decimal_places=2,
        default=Decimal("0.00"),
        validators=[MinValueValidator(Decimal("0.00"))],
    )
    due_at = models.DateTimeField(null=True, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["code"]
        indexes = [models.Index(fields=["owner", "status"])]

    def __str__(self):
        return f"{self.code}: {self.title}"

import csv
import io
from datetime import UTC, datetime

from django.core.files.base import ContentFile
from django.core.files.storage import default_storage

from .models import Card


def due_at_utc(day):
    """The ops team measures its end-of-day deadline at 17:00 UTC."""
    return datetime(day.year, day.month, day.day, 17, tzinfo=UTC)


def cards_for(user, status=None):
    cards = Card.objects.filter(owner=user)
    if status in Card.Status.values:
        cards = cards.filter(status=status)
    return cards


def export_cards(user):
    output = io.StringIO(newline="")
    writer = csv.writer(output)
    writer.writerow(["code", "title", "status", "estimate"])
    for card in cards_for(user):
        # Prevent spreadsheet software interpreting user content as formulas.
        title = "'" + card.title if card.title.startswith(("=", "+", "-", "@")) else card.title
        writer.writerow([card.code, title, card.status, str(card.estimate)])
    return default_storage.save(f"queue-{user.pk}.csv", ContentFile(output.getvalue().encode()))

from decimal import Decimal

from django.contrib.auth import get_user_model
from django.core.exceptions import ValidationError
from django.test import TestCase

from workboard.forms import CardFormSet
from workboard.models import Card
from workboard.services import cards_for


class CardTests(TestCase):
    @classmethod
    def setUpTestData(cls):
        cls.alice = get_user_model().objects.create_user("alice")
        cls.bob = get_user_model().objects.create_user("bob")
        Card.objects.create(code="OPS-002", owner=cls.alice, title="Renew certificate")
        Card.objects.create(code="OPS-001", owner=cls.alice, title="Rotate runner", status="active")
        Card.objects.create(code="PRIVATE-001", owner=cls.bob, title="Private task")

    def test_queue_orders_cards_and_limits_owner(self):
        self.assertQuerySetEqual(
            cards_for(self.alice), ["OPS-001", "OPS-002"], transform=lambda card: card.code
        )

    def test_status_filter_returns_only_matching_work(self):
        self.assertQuerySetEqual(
            cards_for(self.alice, "active"), ["OPS-001"], transform=lambda card: card.code
        )

    def test_unknown_filter_does_not_hide_work(self):
        self.assertEqual(cards_for(self.alice, "invalid").count(), 2)

    def test_display_name_contains_tracking_code(self):
        self.assertEqual(str(Card.objects.get(pk="OPS-002")), "OPS-002: Renew certificate")

    def test_negative_budget_is_invalid(self):
        card = Card(code="NEW", owner=self.alice, title="Budget", estimate=Decimal("-1.00"))
        with self.assertRaises(ValidationError):
            card.full_clean()

    def test_natural_primary_key_is_editable_in_real_formset(self):
        formset = CardFormSet(queryset=cards_for(self.alice))
        self.assertIn("code", formset.forms[0].fields)
        self.assertFalse(Card._meta.pk.auto_created)
        self.assertTrue(Card._meta.pk.editable)

from datetime import UTC, date
from pathlib import Path
from tempfile import TemporaryDirectory

from django.contrib.auth import get_user_model
from django.core.files.storage import default_storage
from django.test import TestCase, override_settings

from workboard.models import Card
from workboard.services import due_at_utc, export_cards


class ServiceTests(TestCase):
    def test_deadline_uses_utc_and_ops_cutoff(self):
        deadline = due_at_utc(date(2026, 11, 3))
        self.assertEqual(deadline.hour, 17)
        self.assertEqual(deadline.tzinfo, UTC)

    def test_exports_use_dedicated_storage_not_general_uploads(self):
        user = get_user_model().objects.create_user("reporter")
        Card.objects.create(code="REP-001", owner=user, title="Report")
        with (
            TemporaryDirectory() as folder,
            override_settings(
                EXPORT_ROOT=folder,
                MEDIA_ROOT=str(Path(folder) / "uploads"),
                STORAGES={
                    "default": {"BACKEND": "workboard.storage.ExportStorage"},
                    "staticfiles": {
                        "BACKEND": "django.contrib.staticfiles.storage.StaticFilesStorage"
                    },
                },
            ),
        ):
            name = export_cards(user)
            self.assertEqual(Path(default_storage.path(name)).parent, Path(folder))
            self.assertEqual(default_storage.url(name), f"/exports/{name}")
            self.assertIn("REP-001", Path(default_storage.path(name)).read_text())

    def test_csv_title_is_escaped_for_spreadsheet_import(self):
        user = get_user_model().objects.create_user("reporter")
        Card.objects.create(code="REP-001", owner=user, title="=1+1")
        with (
            TemporaryDirectory() as folder,
            override_settings(
                EXPORT_ROOT=folder,
                STORAGES={
                    "default": {"BACKEND": "workboard.storage.ExportStorage"},
                    "staticfiles": {
                        "BACKEND": "django.contrib.staticfiles.storage.StaticFilesStorage"
                    },
                },
            ),
        ):
            name = export_cards(user)
            with default_storage.open(name, "rb") as export:
                self.assertIn("'=1+1", export.read().decode())

from datetime import date
from decimal import Decimal
from tempfile import TemporaryDirectory

from django.contrib.auth import get_user_model
from django.test import Client, TestCase, override_settings
from django.urls import reverse
from django.utils import translation

from workboard.models import Card
from workboard.services import due_at_utc


class BoardTests(TestCase):
    def setUp(self):
        self.alice = get_user_model().objects.create_user("alice")
        self.bob = get_user_model().objects.create_user("bob")
        self.own = Card.objects.create(
            code="OPS-001", owner=self.alice, title="Upgrade runner", estimate=Decimal("1234.50")
        )
        self.other = Card.objects.create(code="PRIVATE-001", owner=self.bob, title="Private task")
        self.client.force_login(self.alice)

    def payload(self, **changes):
        return {
            "form-TOTAL_FORMS": "1",
            "form-INITIAL_FORMS": "1",
            "form-MIN_NUM_FORMS": "0",
            "form-MAX_NUM_FORMS": "1000",
            "form-0-code": self.own.code,
            "form-0-title": self.own.title,
            "form-0-status": "open",
            "form-0-estimate": "1234.50",
            **changes,
        }

    def test_anonymous_board_requires_login(self):
        response = Client().get(reverse("board"))
        self.assertRedirects(response, "/accounts/login/?next=/", fetch_redirect_response=False)

    def test_anonymous_bulk_edit_requires_login(self):
        response = Client().post(reverse("edit-cards"), self.payload())
        self.assertEqual(response.status_code, 302)
        self.assertTrue(Card.objects.filter(pk=self.own.pk).exists())

    def test_board_shows_own_work_only(self):
        response = self.client.get(reverse("board"))
        self.assertContains(response, "Upgrade runner")
        self.assertNotContains(response, "Private task")

    def test_empty_queue_has_helpful_message(self):
        Card.objects.filter(owner=self.alice).delete()
        self.assertContains(self.client.get(reverse("board")), "Your queue is clear")

    def test_single_card_heading(self):
        self.assertContains(self.client.get(reverse("board")), "1 work item")

    def test_status_filter_is_used_by_board(self):
        self.own.status = "done"
        self.own.save()
        self.assertContains(
            self.client.get(reverse("board"), {"status": "open"}), "Your queue is clear"
        )

    @override_settings(LANGUAGE_CODE="de")
    def test_budget_stays_machine_readable_in_translated_interface(self):
        with translation.override("de"):
            self.assertContains(self.client.get(reverse("board")), ">1234.50<", html=False)

    def test_bulk_editor_never_lists_another_users_card(self):
        response = self.client.get(reverse("edit-cards"))
        self.assertContains(response, "OPS-001")
        self.assertNotContains(response, "PRIVATE-001")

    def test_bulk_edit_updates_owned_card(self):
        response = self.client.post(
            reverse("edit-cards"), self.payload(**{"form-0-title": "Updated"})
        )
        self.assertRedirects(response, reverse("board"), fetch_redirect_response=False)
        self.own.refresh_from_db()
        self.assertEqual(self.own.title, "Updated")
        self.other.refresh_from_db()
        self.assertEqual(self.other.title, "Private task")

    def test_bulk_delete_deletes_owned_card(self):
        response = self.client.post(reverse("edit-cards"), self.payload(**{"form-0-DELETE": "on"}))
        self.assertEqual(response.status_code, 302)
        self.assertFalse(Card.objects.filter(pk=self.own.pk).exists())
        self.assertTrue(Card.objects.filter(pk=self.other.pk).exists())

    def test_invalid_bulk_edit_leaves_database_unchanged(self):
        response = self.client.post(
            reverse("edit-cards"), self.payload(**{"form-0-estimate": "-1"})
        )
        self.assertEqual(response.status_code, 400)
        self.own.refresh_from_db()
        self.assertEqual(self.own.estimate, Decimal("1234.50"))

    def test_csrf_is_required_for_browser_bulk_edit(self):
        browser = Client(enforce_csrf_checks=True)
        browser.force_login(self.alice)
        self.assertEqual(browser.post(reverse("edit-cards"), self.payload()).status_code, 403)

    def test_schedule_updates_only_open_owned_cards(self):
        done = Card.objects.create(
            code="OPS-DONE", owner=self.alice, title="Finished", status="done"
        )
        response = self.client.post(reverse("schedule-cards"), {"day": "2026-11-03"})
        self.assertEqual(response.status_code, 302)
        self.own.refresh_from_db()
        self.assertEqual(self.own.due_at, due_at_utc(date(2026, 11, 3)))
        done.refresh_from_db()
        self.other.refresh_from_db()
        self.assertIsNone(done.due_at)
        self.assertIsNone(self.other.due_at)

    def test_invalid_schedule_returns_validation_error(self):
        response = self.client.post(reverse("schedule-cards"), {"day": "bad-date"})
        self.assertContains(response, "Choose a valid date", status_code=400)

    def test_schedule_requires_post(self):
        self.assertEqual(self.client.get(reverse("schedule-cards")).status_code, 405)

    def test_export_download_excludes_other_owners(self):
        with (
            TemporaryDirectory() as folder,
            override_settings(
                EXPORT_ROOT=folder,
                STORAGES={
                    "default": {"BACKEND": "workboard.storage.ExportStorage"},
                    "staticfiles": {
                        "BACKEND": "django.contrib.staticfiles.storage.StaticFilesStorage"
                    },
                },
            ),
        ):
            response = self.client.get(reverse("export-cards"))
            self.assertEqual(response.status_code, 200)
            content = b"".join(response.streaming_content).decode()
            response.close()
            self.assertIn("OPS-001,Upgrade runner,open,1234.50", content)
            self.assertNotIn("PRIVATE-001", content)
